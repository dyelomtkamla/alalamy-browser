    for (const k of KEYS) if (k in o) { try { delete o[k]; } catch (e) {} }
    if (o.playerResponse) clean(o.playerResponse);
    if (Array.isArray(o)) for (const x of o) if (x && x.playerResponse) clean(x.playerResponse);

  };

  JSON.parse = function () { const r = jp.apply(this, arguments); try { if (hasAds(r)) clean(r); } catch (e) {} return r; };

  let saved = null, jumped = '';
  const adUI = (p) => !!p.querySelector('.ytp-ad-player-overlay, .ytp-ad-player-overlay-layout, .ytp-ad-preview-container, .ytp-ad-text, .ytp-ad-skip-button-container, .ytp-skip-ad-button, .ytp-ad-persistent-progress-bar-container, .ytp-ad-duration-remaining, .video-ads > *, .ad-interrupting');

    if (inAd) {
      if (!saved) saved = { muted: v.muted, rate: v.playbackRate > 4 ? 1 : (v.playbackRate || 1) };
      v.muted = true;
      try { if (v.playbackRate < 8) v.playbackRate = 8; } catch (e) {}

      if (b) { b.click(); return; }
      const key = (v.currentSrc || v.src || '') + '|' + Math.round(v.duration || 0);

      const separate = isFinite(v.duration) && v.duration > 0 && v.duration <= 75 && (!len || Math.abs(len - v.duration) < 2);
      if (separate && jumped !== key && v.currentTime < v.duration - 1) { jumped = key; v.currentTime = Math.max(0, v.duration - 0.3); }
    } else if (saved) {
      v.muted = saved.muted; try { v.playbackRate = saved.rate; } catch (e) {} saved = null;
    } else if (v.playbackRate > 4) { try { v.playbackRate = 1; } catch (e) {} }

    const p = document.querySelector('#movie_player');
    const v = p && p.querySelector('video');
    if (!v || document.hidden) { stuckSince = 0; return; }
